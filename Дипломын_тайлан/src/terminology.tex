\chapter*{НЭР ТОМЬЁО БА ТОВЧИЛСОН ҮГС}
\addcontentsline{toc}{chapter}{НЭР ТОМЬЁО БА ТОВЧИЛСОН ҮГС}
\begin{sloppy}

\begin{itemize}
    \item \textbf{AR (Augmented Reality):} Бодит орчны дээр дижитал контент давхарлан харуулах технологи.
    \item \textbf{LBS (Location-Based Services):} Хэрэглэгчийн байршилд тулгуурлан контент, үйлчилгээ, мэдэгдэл хүргэх систем.
    \item \textbf{GNSS/GPS:} Дэлхийн хэмжээний навигацийн хиймэл дагуулын систем болон түүний хэрэглэгчийн байршил тодорхойлох дэд систем.
    \item \textbf{IMU (Inertial Measurement Unit):} Акселерометр, гироскоп зэрэг инерцийн мэдрэгчийг нэгтгэсэн модуль.
    \item \textbf{SLAM (Simultaneous Localization and Mapping):} Орчны газрын зураг үүсгэхийн зэрэгцээ төхөөрөмжийн байрлал, чиглэлийг үнэлэх арга.
    \item \textbf{VIO (Visual-Inertial Odometry):} Камерын мэдээлэл ба IMU өгөгдлийг нэгтгэн хөдөлгөөнийг тооцоолох арга.
    \item \textbf{Sensor Fusion:} Олон төрлийн мэдрэгчийн мэдээллийг нэгтгэн илүү найдвартай үнэлгээ гаргах тооцооллын арга.
    \item \textbf{Anchor:} AR орчинд виртуал объектыг тогтвортой бэхлэх орон зайн лавлах цэг.
    \item \textbf{Hysteresis:} Идэвхжих ба идэвхгүйжих босгыг тусад нь авч, төлөв байн байн савлахаас сэргийлэх логик.
    \item \textbf{Reward Claim:} Хэрэглэгч тухайн байршлын шагналыг системд баталгаажуулан авах үйлдэл.
    \item \textbf{Collection Retrieval:} Хэрэглэгчийн цуглуулсан шагналын мэдээллийг backend-ээс татах урсгал.
\end{itemize}

\end{sloppy}
